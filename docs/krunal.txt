const mongoose = require("mongoose");

const markSchema = new mongoose.Schema(
{
    markId: {
        type: String,
        unique: true,
        required: true
    },

    studentId: {
        type: String,
        required: true
    },

    examId: {
        type: String,
        required: true
    },

    classId: {
        type: String,
        required: true
    },

    academicYearId: {
        type: String,
        required: true
    },

    subjects: [
        {
            subjectId: {
                type: String,
                required: true
            },

            subjectTeacherId: {
                type: String,
                required: true
            },

            marks: {
                type: Number,
                required: true,
                min: 0
            },

            grade: {
                type: String,
                required: true
            },

            updatedAt: {
                type: Date,
                default: Date.now
            }
        }
    ],

    totalMarks: {
        type: Number,
        default: 0
    },

    percentage: {
        type: Number,
        default: 0,
        min: 0,
        max: 100
    },

    behavior: {
        type: String,
        default: ""
    },

    remarks: {
        type: String,
        default: ""
    },

    resultStatus: {
        type: String,
        enum: ["DRAFT", "PENDING", "LOCKED", "PUBLISHED"],
        default: "DRAFT"
    },

    lockedBy: {
        type: String,
        default: null
    },

    publishedBy: {
        type: String,
        default: null
    }
},
{
    timestamps: true
}
);

module.exports = mongoose.model("Mark", markSchema);
